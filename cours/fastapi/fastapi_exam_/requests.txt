# Test de l'endpoint GET /verify
curl -X GET "http://localhost:8000/verify" -H "Accept: application/json"

# Test de l'endpoint POST /generate_quiz (Succès)
# ---------------------------------------------------------------------------------------
curl -X POST "http://localhost:8000/generate_quiz" \
     -H "Accept: application/json" \
     -H "Content-Type: application/json" \
     -H "Authorization: Basic $(echo -n 'bob:builder' | base64)" \
     -d '{
           "test_type": "Test de positionnement",
           "categories": ["BDD", "Docker"],
           "number_of_questions": 5
         }'

# Test de l'endpoint POST /generate_quiz (Authentification Invalide - 401 Unauthorized)
